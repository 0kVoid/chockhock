local function CreateCSModel(mdl)
    local model = ClientsideModel(mdl)
    model:SetNoDraw(true)
    return model
end

local function UpdateCSModel(ply, model, tbl)
    if not ply or not model or not tbl then return end
    
    local mdl = model
    local playerModel = ply:GetModel()
    local layers = tbl.layers or {}
    
    for i = 0, 13 do
        if mdl:IsValidLayer(i) then
            local l = layers[i]
            if l then
                mdl:SetLayerCycle(l.cycle)
                mdl:SetLayerSequence(l.sequence)
                mdl:SetLayerWeight(l.weight)
            end
        end
    end
    
    mdl:SetSequence(tbl.sequence or 0)
    mdl:SetCycle(tbl.cycle or 0)
    mdl:SetPoseParameter("aim_pitch", tbl.angles.p or 0)
    mdl:SetPoseParameter("head_pitch", 0)
    mdl:SetPoseParameter("body_yaw", tbl.angles.y or 0)
    mdl:SetPoseParameter("aim_yaw", 0)
    mdl:SetPoseParameter("move_x", tbl.movex or 0)
    mdl:SetPoseParameter("move_y", tbl.movey or 0)
    mdl:SetAngles(Angle(0, tbl.angles.y or 0, 0))
    mdl:SetModel(playerModel)
    mdl:SetPos(tbl.origin)
end

local function DrawRealChams()
    if not vars.aa_fake_chams then return end
    if not me:Alive() then return end
    
    local mymodel = me:GetModel()
    
    if newModel ~= mymodel then
        if realModel then realModel:Remove() end
        if fakeModel then fakeModel:Remove() end
        realModel = CreateCSModel(mymodel)
        fakeModel = CreateCSModel(mymodel)
        newModel = mymodel
    end
    
    if not realModel or not fakeModel then
        realModel = CreateCSModel(mymodel)
        fakeModel = CreateCSModel(mymodel)
    end
    
    local realTbl = {
        layers = {},
        angles = Angle(pitch, realAngles.y, 0),
        sequence = me:GetSequence(),
        cycle = me:GetCycle(),
        origin = me:GetPos(),
        movex = me:GetPoseParameter("move_x"),
        movey = me:GetPoseParameter("move_y"),
    }
    
    for i = 0, 13 do
        if me:IsValidLayer(i) then
            realTbl.layers[i] = {
                cycle = me:GetLayerCycle(i),
                sequence = me:GetLayerSequence(i),
                weight = me:GetLayerWeight(i)
            }
        end
    end
    
    local fakeTbl = {
        layers = realTbl.layers,
        angles = Angle(pitch, fakeAngles.y, 0),
        sequence = realTbl.sequence,
        cycle = realTbl.cycle,
        origin = realTbl.origin,
        movex = realTbl.movex,
        movey = realTbl.movey,
    }
    
    UpdateCSModel(me, realModel, realTbl)
    render.SuppressEngineLighting(true)
    render.MaterialOverride(selfillum)
    render.SetColorModulation(0, 1, 0)
    render.SetBlend(1)
    realModel:SetRenderMode(1)
    realModel:DrawModel()
    
    UpdateCSModel(me, fakeModel, fakeTbl)
    render.MaterialOverride(selfillum)
    render.SetColorModulation(1, 0, 0)
    fakeModel:SetRenderMode(1)
    fakeModel:DrawModel()
    
    render.MaterialOverride()
    render.SetColorModulation(1, 1, 1)
    render.SetBlend(1)
    render.SuppressEngineLighting(false)
end

local function IsOutOfFOV(ent)
    if me:GetObserverMode() == 0 then
        local Disp = (ent:GetPos() - me:GetShootPos()):GetNormalized()
        return Disp:Dot(me:EyeAngles():Forward()) < math_cos(math_atan(ent:BoundingRadius() / (ent:GetPos():Distance(me:GetShootPos()))) + math_rad(56))
    end
end

local function combobox(name, val, x, y, choices, mappings, parent)
    local combo = vCreate("DComboBox", parent)
    combo:SetPos(x, y)
    combo:SetSize(140, 20)
    combo:SetSortItems(false)
    combo:SetValue(choices[1])
    combo:SetTextColor(colors.white_menu)

    for _, choice in ip(choices) do
        combo:AddChoice(choice)
    end

    function combo:OnSelect(index, mode)
        vars[val] = mappings and mappings[mode] or mode
    end

    combo.Paint = function(self, w, h)
        sSetDrawColor(colors.hover)
        sDrawRect(0, 0, w, h)
        sDrawOutlinedRect(0, 0, w, h)
    end

    combo.DropButton.Paint = function(self, w, h)
        sSetDrawColor(colors.hover)
        sDrawRect(0, 0, w, h)
    end

    combo.OnMenuOpened = function(pnl)
        local menu = pnl.Menu
        if IsValid(menu) then
            menu.Paint = function(self, w, h)
                sSetDrawColor(colors.hover)
                sDrawRect(0, 0, w, h)
            end
            for _, option in ip(menu:GetCanvas():GetChildren()) do
                option:SetTextColor(colors.white_menu)
            end
        end
    end
end

local function normalize_yaw(yaw)
    if yaw > 180 or yaw < -180 then
        local revolutions = math_round(math_abs(yaw / 360))
        
        if yaw < 0 then
            yaw = yaw + 360 * revolutions
        else
            yaw = yaw - 360 * revolutions
        end
    end
    
    return yaw
end

local function MovementFix(cmd, wish_yaw)
    local pitch = math_NormAngle(cmd:GetViewAngles().x)
    local inverted = -1

    if (pitch > 89 or pitch < -89) then
        inverted = 1
    end

    local ang_diff = math_rad(math_NormAngle((cmd:GetViewAngles().y - wish_yaw) * inverted))
    local forwardmove = cmd:GetForwardMove()
    local sidemove = cmd:GetSideMove()

    local new_forwardmove = forwardmove * -math_cos(ang_diff) * inverted + sidemove * math_sin(ang_diff)
    local new_sidemove = forwardmove * math_sin(ang_diff) * inverted + sidemove * math_cos(ang_diff)

    cmd:SetForwardMove(new_forwardmove)
    cmd:SetSideMove(new_sidemove)
end

end

    if me:Crouching() then
        cone = cone * 0.85
    end

    if not isnumber(cone) then return ang end
    Dir = Ang(math_rand(cone, -cone), math_rand(cone, -cone), 0) * 25

    clumpSpread = clumpSpread or activeWeapon.ClumpSpread
    if not isnumber(clumpSpread) then clumpSpread = 0 end

    Dir2 = Dir

    end

    if not isangle(ang) or not isangle(Dir2) then return ang end

    return ang - me:GetViewPunchAngles() + Dir2

function CustomSpread.tfa(cmd, ang) -- WORK
    if not vars or not activeWeapon or not IsValid(activeWeapon) then return ang end
    local wep = activeWeapon
    if not isfunction(wep.CalculateConeRecoil) or not isfunction(wep.ComputeBulletDeviation) then return ang end

    local con, _ = wep:CalculateConeRecoil()
    if not con or not wep:CanPrimaryAttack() then return ang end

    local dYaw, dPitch = wep:ComputeBulletDeviation(1, 1, con)
    if not dYaw or not dPitch then return ang end

    local aimAng = Ang(ang)
    local right, up = aimAng:Right(), aimAng:Up()

    aimAng:RotateAroundAxis(up, -dYaw)
    aimAng:RotateAroundAxis(right, -dPitch)

    return aimAng
end

local function Spread(cmd, ang, spread)
    if not isangle(ang) then return ang end
    if not isvector(spread) then return ang end

    local ok, dir = pcall(ded.PredictSpread, cmd, spread)
    if not ok or not isvector(dir) then return ang end

    local spreadangle = ang + dir:Angle()
    spreadangle:Normalize()
    return spreadangle
end

local function NoSpread(cmd, ang)
    local class = activeWeaponClass or ""
    if class == "" then return ang end
    if not isangle(ang) then return ang end

    local wep = activeWeapon
    if not IsValid(wep) then return ang end

    local base = string.Split(class, "_")[1] or ""

    if wep.IsTFAWeapon then
        base = "tfa"
    elseif wep.Base == "cw_base" then
        base = "cw"
    elseif wep.ARC9Weapon then
        base = "arc9"
    elseif wep.ARCWeapon then
        base = "arccw"
    end

    if SpreadComps[base] then
        local ok, result = pcall(SpreadComps[base], cmd, ang)
        if ok and isangle(result) then
            return result
        end
        return ang
    end

    if isvector(cones[class]) then
        return Spread(cmd, ang, cones[class])
    end

    return ang
end

    elseif wep and wep.IsTFAWeapon then

            ang.p = ang.p - wep:GetViewPunchP()

        end
    else

        end
    end

    return ang
end

        if ammoType == "7.62x54MMR" or ammoType == ".338 Lapua" or ammoType == ".50 AE" then
            damageMulti = 1
        elseif tr.MatType == MAT_CONCRETE or tr.MatType == MAT_METAL then
            damageMulti = 0.3
        elseif tr.MatType == MAT_WOOD or tr.MatType == MAT_PLASTIC or tr.MatType == MAT_GLASS then
            damageMulti = 0.8
        elseif tr.MatType == MAT_FLESH or tr.MatType == MAT_ALIENFLESH then
            damageMulti = 0.9
        end

        if penTrace.MatType == MAT_GLASS then
            bounceNum = bounceNum - 1
        end

        return BulletPenetrate(penTrace, bounceNum + 1, damage * damageMulti)
    end

    local firstTrace = util.TraceLine(traceStruct)
    return BulletPenetrate(firstTrace, 0, primaryDamage)
end

            end
        end
    end

local arccwPenetrationTable = {
    [MAT_GLASS] = 2.0,
    [MAT_PLASTIC] = 1.5,
    [MAT_WOOD] = 1.0,
    [MAT_FLESH] = 2.0,
    [MAT_ALIENFLESH] = 2.0,
    [MAT_METAL] = 0.5,
    [MAT_CONCRETE] = 0.4,
    [MAT_DIRT] = 0.8,
    [MAT_SAND] = 0.7,

CustomWall.arccw = function(targetPos, plyTarget)
    if not IsValid(me) then return false end

    local eyePos = me:EyePos()
    local dir = (targetPos - eyePos):GetNormalized()
    local weapon = me:GetActiveWeapon()
    if not IsValid(weapon) then return false end
    local penetration = weapon:GetBuff("Penetration") or 0
    if penetration <= 0 then return false end
    local damage = weapon:GetDamage(100, true) or 0
    if damage <= 0 then return false end

    local function BulletPenetrate(tr, penLeft, dmg, depth)
        if depth > 10 then return false end
        if not tr then return false end
        if penLeft <= 0 then return false end
        if dmg <= 0 then return false end

        if plyTarget and tr.Entity == plyTarget then
            if IGNORE_HEAD_UNHITABLE then
                return tr.HitGroup == HITGROUP_HEAD
            else
                return true
            end
        end

        if tr.HitSky then return false end
        if tr.Fraction >= 1 then return false end

        local penMult = arccwPenetrationTable[tr.MatType] or 0.5

        local dmgMult = 0.5
        if tr.MatType == MAT_CONCRETE or tr.MatType == MAT_METAL then
            dmgMult = 0.3
        elseif tr.MatType == MAT_WOOD or tr.MatType == MAT_PLASTIC or tr.MatType == MAT_GLASS then
            dmgMult = 0.8
        elseif tr.MatType == MAT_FLESH or tr.MatType == MAT_ALIENFLESH then
            dmgMult = 0.9
        end

        local penCost = 1 / penMult
        local newPenLeft = penLeft - penCost
        if newPenLeft <= 0 then return false end
		local traceStruct = {
            start = tr.HitPos + dir * 0.1,
            endpos = tr.HitPos + dir * 32768,
            mask = MASK_SHOT,
            filter = me
        }

        local penTrace = util.TraceLine(traceStruct)

        if plyTarget and penTrace.Entity == plyTarget then
            local finalDamage = dmg * dmgMult
            if finalDamage < 5 then return false end
            return true
        end

        return BulletPenetrate(penTrace, newPenLeft, dmg * dmgMult, depth + 1)
    end

    local traceStruct = {
        start = eyePos,
        endpos = eyePos + dir * 32768,
        mask = MASK_SHOT,
        filter = me
    }

    local firstTrace = util.TraceLine(traceStruct)
    return BulletPenetrate(firstTrace, penetration, damage, 0)
end

local function AutoWall(targetPos, plyTarget)
    if not (activeWeaponClass and activeWeapon and IsValid(activeWeapon)) then return false end

    local base
    if activeWeapon.Base == "cw_base" then
        base = "cw"
    elseif activeWeapon.ArcCW then
        base = "arccw"
    else
        base = string.Split(activeWeaponClass, "_")[1]
    end
    
    local wallFunc = CustomWall[base]
    if wallFunc then
        local ok, result = pcall(wallFunc, targetPos, plyTarget)
        if ok then
            return result
        else
            print("[AutoWall] Error in CustomWall[" .. base .. "]: " .. tostring(result))
            return false
        end
    end

    local tr = util.TraceLine({
        start  = me:EyePos(),
        endpos = targetPos,
        mask   = MASK_SHOT,
        filter = me
    })
    return tr.Entity == plyTarget or tr.Fraction == 1
end

local function GetAngleDifference(from, to)
    local fromF, toF = from:Forward(), to:Forward()
    return math_deg(math_acos(toF:Dot(fromF)))
end

local function GetBestTarget()
    local bestTarget = nil
    local bestFov = vars.aim_fov
	
	

    for _, ply in ip(player_GetAll()) do
        if ply ~= me and ply:Alive() and !ply:IsDormant() then
            local boneIndex = ply:LookupBone(vars.boneAim)
            if boneIndex then
                local bonePos = ply:GetBonePosition(boneIndex)
                local steamID = ply:SteamID()
                local god = ply:IsFlagSet(FL_GODMODE)
                local inVeh = ply:InVehicle()
                if bonePos and VisibleCheck(ply, bonePos) and !IsFriend(steamID) and !god and !inVeh then
                    local targetDirection = (bonePos - me:GetShootPos()):Angle()
                    targetDirection:Normalize()

                    local angDiff = math_abs(math_NormAngle(GetAngleDifference(fa, targetDirection)))

                    if angDiff < bestFov then
                        bestFov = angDiff
                        bestTarget = ply
                    end
                end
            end
        end
    end

    return bestTarget
end

local function predictTarget(pos,trg)
    if !pos or not trg then return nulVec end
    local lvel,tvel,frm,eng = me:GetAbsVelocity(), trg:GetAbsVelocity(), realFrameTime(), eTickInter
    if vars.predtype == 'none' then return pos end
    if vars.predtype == 'velocity' then return pos+((lvel-tvel)*(frm/(1/eng))) end
    if vars.predtype == 'engine'  then return tvel == nulVec and pos or pos+tvel*eng*frm-lvel*eng end
end


local function CrossbowPred(cmd)
    if not vars.aim_crossbow_pred then return end
    
    local wep = activeWeapon
    if not IsValid(wep) then return end
    
    local class = toStr(activeWeaponClass)
    if class ~= "weapon_crossbow" and not string.find(class, "crossbow") then return end
    
    local target = GetBestTarget()
    if not target then return end
    
    local eyePos = me:EyePos()
    
    local targetPos = target:GetPos()
    local targetCenter = target:OBBCenter()
    
    local aimPos = targetPos + targetCenter
    local aimAngle = (aimPos - eyePos):Angle()
    local aimVector = aimAngle:Forward()
    
    local boltSpeed = vars.aim_crossbow_speed
    local distance = eyePos:Distance(aimPos)
    local travelTime = distance / boltSpeed
    
    -- Movement simulation
    local predTime = (ded.GetLatency(0) + ded.GetLatency(1)) + travelTime
    
    if predTime > 1 then return end
    
    ded.StartSimulation(target:EntIndex())
    
    for i = 1, math.floor(predTime / eTickInter) do
        ded.SimulateTick()
    end
    
    local data = ded.GetSimulationData()
    
    aimPos = data.m_vecAbsOrigin + targetCenter
    distance = eyePos:Distance(aimPos)
    travelTime = distance / boltSpeed
    
    ded.FinishSimulation()
    
    -- Gravity simulation
    local gravity = GetConVar("sv_gravity"):GetFloat() * 0.05
    gravity = (gravity * eTickInter) * math.floor(travelTime / eTickInter)
    
    aimPos.z = aimPos.z + gravity
    
    -- Aimbot
    local finalVec = Vector(aimPos.x, aimPos.y, aimPos.z)
    local finalAng = (finalVec - eyePos):Angle()
    finalAng:Normalize()
    
    cmd:SetViewAngles(finalAng)
    
    return true
end

local get_velocity_degree = function(velocity)
    local tmp = math_deg(math_atan(30.0 / velocity))

    if (tmp > 90.0) then
        return 90.0
    elseif (tmp < 0.0) then
        return 0.0
    else
        return tmp
    end
end

local old_yaw = 0.0
local M_RADPI = 57.295779513082
local side_speed = 10000
local function Strafe(cmd)
    local velocity = me:GetVelocity()
    velocity.z = 0.0

    local forwardmove = cmd:GetForwardMove()
    local sidemove = cmd:GetSideMove()

    if (!forwardmove || !sidemove) then
        return
    end

    if(velocity:Length2D() <= 15.0 && !(forwardmove != 0 || sidemove != 0)) then
        return
    end

    local flip = cmd:TickCount() % 2 == 0

    local turn_direction_modifier = flip && 1.0 || -1.0
    local viewangles = Ang(fa.x, fa.y, fa.z)

    if (forwardmove || sidemove) then
        cmd:SetForwardMove(0)
        cmd:SetSideMove(0)

        local turn_angle = math_atan2(-sidemove, forwardmove)
        viewangles.y = viewangles.y + (turn_angle * M_RADPI)
    elseif (forwardmove) then
        cmd:SetForwardMove(0)
    end

    local strafe_angle = math_deg(math_atan(15 / velocity:Length2D()))

    if (strafe_angle > 90) then
        strafe_angle = 90
    elseif (strafe_angle < 0) then
        strafe_angle = 0
    end

    local temp = Vec(0, viewangles.y - old_yaw, 0)
    temp.y = math_NormAngle(temp.y)

    local yaw_delta = temp.y
    old_yaw = viewangles.y

    local abs_yaw_delta = math_abs(yaw_delta)

    if (abs_yaw_delta <= strafe_angle || abs_yaw_delta >= 30) then
        local velocity_angles = velocity:Angle()

        temp = Vec(0, viewangles.y - velocity_angles.y, 0)
        temp.y = math_NormAngle(temp.y)

        local velocityangle_yawdelta = temp.y
        local velocity_degree = get_velocity_degree(velocity:Length2D() * 128)

        if (velocityangle_yawdelta <= velocity_degree || velocity:Length2D() <= 15) then
            if (-velocity_degree <= velocityangle_yawdelta || velocity:Length2D() <= 15) then
                viewangles.y = viewangles.y + (strafe_angle * turn_direction_modifier)
                cmd:SetSideMove(side_speed * turn_direction_modifier)
            else
                viewangles.y = velocity_angles.y - velocity_degree
                cmd:SetSideMove(side_speed)
            end
        else
            viewangles.y = velocity_angles.y + velocity_degree
            cmd:SetSideMove(-side_speed)
        end
    elseif (yaw_delta > 0) then
        cmd:SetSideMove(-side_speed)
    elseif (yaw_delta < 0) then
        cmd:SetSideMove(side_speed)
    end

    local move = Vec(cmd:GetForwardMove(), cmd:GetSideMove(), 0)
    local speed = move:Length()

    local angles_move = move:Angle()

    local normalized_x = math_modf(fa.x + 180, 360) - 180
    local normalized_y = math_modf(fa.y + 180, 360) - 180

    local yaw = math_rad(normalized_y - viewangles.y + angles_move.y)

    if (normalized_x >= 90 || normalized_x <= -90 || fa.x >= 90 && fa.x <= 200 || fa.x <= -90 && fa.x <= 200) then
        cmd:SetForwardMove(-math_cos(yaw) * speed)
    else
        cmd:SetForwardMove(math_cos(yaw) * speed)
    end

    cmd:SetSideMove(math_sin(yaw) * speed)
end

    if not IsValid(v) or v == me or not v:Alive() or v:IsDormant() or me:GetPos():Distance(v:GetPos()) > 10000 then return end
    if not me:InVehicle() and not vars.antiaim and IsOutOfFOV(v) then return end

    local teamColor = team.GetColor(v:Team())
    local r, g, b = teamColor.r / 255, teamColor.g / 255, teamColor.b / 255

    local lightColors = {
        {r * 0.2 + 0.05, g * 0.1 + 0.05, b * 0.3 + 0.2}, -- BOX_FRONT 
        {r * 0.4, g * 0.3 - 0.05, b * 0.2 + 0.1},       -- BOX_BACK 
        {r * 0.7, g * 0.6 - 0.05, b * 0.2 - 0.1},      -- BOX_RIGHT
        {r * 0.8, g * 0.5, b * 0.3 - 0.2},            -- BOX_LEFT
        {r * 0.9, g * 0.8, b * 0.3 - 0.1},            -- BOX_TOP
        {r * 0.4, g * 0.3, b * 0.5 + 0.2}             -- BOX_BOTTOM
    }

    local GodAndNoclip = {
        {0.6 + 0.05, 0.6 + 0.05, 0.6 + 0.05}, -- BOX_FRONT (gris clair)
        {0.7 + 0.1,  0.7 + 0.1,  0.7 + 0.1},  -- BOX_BACK
        {0.8 - 0.05, 0.8 - 0.05, 0.8 - 0.05}, -- BOX_RIGHT
        {0.9,        0.9,        0.9},        -- BOX_LEFT
        {1.0,        1.0,        1.0},        -- BOX_TOP (le plus blanc)
        {0.75,       0.75,       0.75}        -- BOX_BOTTOM
    }

    local isGod = v:IsFlagSet( FL_GODMODE )
    local isNoclipped = v:GetMoveType() == MOVETYPE_NOCLIP
    local colorTable = (isGod or isNoclipped) and GodAndNoclip or lightColors

    rSupEngLight(true)

    rMatOver(selfillum)
    if IsGod and isNoclipped then
        rBlend(0.3)
    end
    for i, color in ip(colorTable) do
        rModelLight(i - 1, unpack(color))
    end
    v:DrawModel()

    rMatOver()
    rColorMod(1, 1, 1)
    rSupEngLight(false)
end

local function _PDE()
	if _G.HideCheatVisuals then return end
    camignorez(true)
    for k,v in ip(player_GetAll()) do
        if (isValid(v) and v:Alive() and v ~= me) then
            if vars.esp_chams then chamz(v) end
        end
    end
    camignorez(false)
end

local tmpVec = Vec(0, 0, 0)

local function drawBox(x, y, w, h, color)
    sSetDrawColor(colors.black)
    sDrawOutlinedRect(x - 1, y - 1, w + 2, h + 2)
    sSetDrawColor(color)
    sDrawOutlinedRect(x, y, w, h)
end

local function boxcorners(v)
    if not IsValid(v) or not v:IsPlayer() then return end

    local min, max = v:GetCollisionBounds()
    local pos = v:GetPos()
    tmpVec.x, tmpVec.y, tmpVec.z = 0, 0, max.z
    local pos2 = pos + tmpVec
    pos, pos2 = tScreen(pos), tScreen(pos2)

    local h = pos.y - pos2.y
    local w = h / 2
    local x = pos.x - w / 2
    local y = pos.y - h

    drawBox(x, y, w, h, GetEntityColor(v))

    -- Santé
    local health = math_clamp(v:Health(), 0, 100)
    local healthFrac = health * 0.01
    local healthBarHeight = h * healthFrac

    sSetDrawColor(colors.black)
    sDrawRect(x - 6, y - 1, 4, h + 2)
    sSetDrawColor(colors.green)
    sDrawRect(x - 5, y + h - healthBarHeight, 2, healthBarHeight)

    -- Armure
    local armor = math_clamp(v:Armor(), 0, 100)
    local armorFrac = armor * 0.01
    local armorBarHeight = h * armorFrac

    sSetDrawColor(colors.black)
    sDrawRect(x + w + 2, y - 1, 4, h + 2)
    sSetDrawColor(colors.armor)
    sDrawRect(x + w + 3, y + h - armorBarHeight, 2, armorBarHeight)
end


local function txtesp(v)
    local pos = tScreen((v:GetPos() + Vec(0, 0, 75)))
    
    local pName = v:Nick()
    if v:IsAdmin() then
        pName = pName .. " [STAFF]"
    end

    dSimpleTextOutlined(pName, vars.cmdfont, pos.x, pos.y, GetEntityColor(v), TEXT_ALIGN_CENTER, TEXT_ALIGN_CENTER, 1, colors.Background1)
end

local function xhair()
    local X, Y = sw * 0.5, sh * 0.5

    if vars.thirdPersonEnabled then
        local trace = util.TraceLine({
            start = me:EyePos(),
            endpos = me:EyePos() + fa:Forward() * 10000,
            filter = me
        })

        if trace.Hit then
            local screenPos = tScreen(trace.HitPos)
            X, Y = screenPos.x or X, screenPos.y or Y
        end
    end

    local size, thickness = 3.5, 2
    sSetDrawColor(255, 0, 0, 255)
    sDrawRect(X - size - thickness, Y - thickness * 0.5, size, thickness)
    sDrawRect(X + thickness, Y - thickness * 0.5, size, thickness)
    sDrawRect(X - thickness * 0.5, Y - size - thickness, thickness, size)
    sDrawRect(X - thickness * 0.5, Y + thickness, thickness, size)
end

local function watermark()
    local padding = 15
    local lineHeight = 16
    local boxWidth = 220
    local boxHeight = 100

    sSetDrawColor(25, 25, 30, 240)
    sDrawRect(10, 10, boxWidth, boxHeight)

    sSetDrawColor(200, 50, 50, 255)
    sDrawRect(10, 10, boxWidth, 2)

    sSetDrawColor(40, 40, 45, 180)
    sDrawRect(10, 30, boxWidth, 1)

    local y = 15
    dDrawText("CHOKCHOK", vars.cmdfont, padding, y, Color(255, 255, 255, 255))

    y = y + lineHeight + 8
    dDrawText("FPS: " .. math_round(1 / FrameTime()), vars.cmdfont, padding, y, Color(200, 200, 200, 255))

    y = y + lineHeight
    dDrawText("Tick: " .. math_round(1 / eTickInter), vars.cmdfont, padding, y, Color(200, 200, 200, 255))

    y = y + lineHeight
    local choke = string.format("%.2f", ded.GetAvgChoke(0) / 100)
    dDrawText("Choke: " .. choke .. "s", vars.cmdfont, padding, y, Color(200, 200, 200, 255))

    y = y + lineHeight
    local ping = math_round((ded.GetLatency(0) + ded.GetLatency(1)) * 1000)
    dDrawText("Ping: " .. ping .. "ms", vars.cmdfont, padding, y, Color(200, 200, 200, 255))
end

local function FC()
    if not vars.aim_aimbot then return end
    
    local X1 = math_tan(math_rad(vars.aim_fov) / 1.25)
    local X2 = math_tan(math_rad(me:GetFOV() / 2))
    local r = X1 / X2
    local radius = r * (sw / 2)
    
    
    local segments = 64
    local centerX, centerY = sw / 2, sh / 2
    
    sSetDrawColor(255, 0, 0, 255)
    
    for i = 0, segments - 1 do
        local angle1 = (i / segments) * math_pi * 2
        local angle2 = ((i + 1) / segments) * math_pi * 2
        
        local x1 = centerX + math_cos(angle1) * radius
        local y1 = centerY + math_sin(angle1) * radius
        local x2 = centerX + math_cos(angle2) * radius
        local y2 = centerY + math_sin(angle2) * radius
        
        sDrawLine(x1, y1, x2, y2)
    end
end

local function xhair()
	if ss then return end
	if _G.HideCheatVisuals then return end
    local X, Y = sw * 0.5, sh * 0.5

    if vars.thirdPersonEnabled then
        local trace = util.TraceLine({
            start = me:EyePos(),
            endpos = me:EyePos() + fa:Forward() * 10000,
            filter = me
        })

        if trace.Hit then
            local screenPos = tScreen(trace.HitPos)
            X, Y = screenPos.x or X, screenPos.y or Y
        end
    end

    local rotation_angle = (CurTime() * 180) % 360
    local scale = 10
    local rad = math.rad

    local points = {
        scale * math.sin(rad(rotation_angle - 90)),
        scale * math.cos(rad(rotation_angle - 90)),
        scale * math.cos(rad(rotation_angle - 90) + 1),

        scale * math.sin(rad(rotation_angle)),
        scale * math.cos(rad(rotation_angle)),
        scale * math.sin(rad(rotation_angle) + 1),

        scale * math.sin(rad(rotation_angle + 90)),
        scale * math.cos(rad(rotation_angle + 90)),
        scale * math.cos(rad(rotation_angle + 90) + 1),

        scale * math.sin(rad(rotation_angle + 180)),
        scale * math.cos(rad(rotation_angle + 180)),
        scale * math.sin(rad(rotation_angle + 180) + 1),
    }

    sSetDrawColor(0, 0, 0, 255)
    for offset_x = -1, 1 do
        for offset_y = -1, 1 do
            if offset_x ~= 0 or offset_y ~= 0 then
                local ox, oy = X + offset_x, Y + offset_y

                sDrawLine(ox, oy, ox + points[1], oy - points[2])
                sDrawLine(ox + points[1], oy - points[2], ox + points[1], oy - points[3])

                sDrawLine(ox, oy, ox + points[4], oy - points[5])
                sDrawLine(ox + points[4], oy - points[5], ox + points[6], oy - points[5])

                sDrawLine(ox, oy, ox + points[7], oy - points[8])
                sDrawLine(ox + points[7], oy - points[8], ox + points[7], oy - points[9])

                sDrawLine(ox, oy, ox + points[10], oy - points[11])
                sDrawLine(ox + points[10], oy - points[11], ox + points[12], oy - points[11])
            end
        end
    end

    sSetDrawColor(255, 0, 0, 255)

    sDrawLine(X, Y, X + points[1], Y - points[2])
    sDrawLine(X + points[1], Y - points[2], X + points[1], Y - points[3])

    sDrawLine(X, Y, X + points[4], Y - points[5])
    sDrawLine(X + points[4], Y - points[5], X + points[6], Y - points[5])

    sDrawLine(X, Y, X + points[7], Y - points[8])
    sDrawLine(X + points[7], Y - points[8], X + points[7], Y - points[9])

    sDrawLine(X, Y, X + points[10], Y - points[11])
    sDrawLine(X + points[10], Y - points[11], X + points[12], Y - points[11])
end


local function _HP()
    if ss then return end
	if _G.HideCheatVisuals then return end
    if vars.unload then
        for k,v in ip(player_GetAll()) do
            if not IsValid(v) or v == me or not v:Alive() or v:IsDormant() then continue end
            if me:GetPos():Distance(v:GetPos()) > 10000 then continue end
            if not me:InVehicle() and not vars.antiaim and IsOutOfFOV(v) then continue end

            if vars.esp_boxes then boxcorners(v) end
            if vars.esp then txtesp(v) end
        end
        if vars.aim_aimbot and !vars.thirdPersonEnabled then
            FC()
            if vars.target and shouldFire() then
                local bonePos = getCenterBone(vars.target)
                if not bonePos then return end
                local pos = tScreen(bonePos)
                sSetDrawColor( colors.red )
                sDrawLine( pos.x, pos.y, sw / 2, sh / 2 )
            end
        end
        xhair()
        watermark()
     end
end

            sSetDrawColor( colors.red )

        cam.End3D2D()

        camignorez( false )
    end
end
